# Transcrypt Web Engine — минимальный набор зависимостей для engine_server.py
# Без PyQt6, без переводчиков, без PDF — только то, что нужно веб-интерфейсу.
#
# Установочный скрипт скачивает этот файл и engine_server.py из GitHub
# в ~/.transcrypt-engine/, создаёт venv и запускает движок на 127.0.0.1:8765.

# ВАЖНО: openai-whisper использует pkg_resources в своём setup.py (старые версии),
# а pkg_resources удалён в setuptools 81+. Фиксируем setuptools<81 чтобы
# избежать ModuleNotFoundError при установке на Python 3.12+.
# Подробнее: https://setuptools.pypa.io/en/latest/history.html#v81-0-0
setuptools<81

# Whisper + зависимости (новая версия 20250625 — без pkg_resources в setup.py)
openai-whisper==20250625

# Скачивание медиа с YouTube/TikTok/Instagram с обходом блокировок.
# 2026.8.19 — проверенный минимум; КОМЕНТАРИЙ ПРО ПИНЫ: выше пин не ставим —
# yt-dlp можно обновлять кнопкой «Обновить yt-dlp» в Настройках сайта
# (движок ≥ 1.3.6), а при повторном запуске установщика pip install -r
# с жёстким пином ОТКАТИЛ бы такое обновление. >= не мешает ни тому, ни другому.
# При жалобах «yt-dlp не получил информацию» сначала обновляйте yt-dlp кнопкой.
#
# [default] ОБЯЗАТЕЛЕН (движок ≥ 1.3.8): ставит yt-dlp-ejs — скрипты решателя
# JS-challenge (n-challenge/PO Token) локально из pip. Без них решатель
# скачивает скрипты с GitHub в рантайме, что ломается при блокировках.
#
# deno ОБЯЗАТЕЛЕН (движок ≥ 1.3.8): JS-рантайм для решателя EJS, wheels
# для Windows/macOS/Linux (бинарник кладётся в venv). Без рантайма yt-dlp
# не может решать вызовы YouTube → «... require a GVS PO Token ... skipped»
# → «Requested format is not available» на ВСЕХ клиентах. Именно поэтому
# скачивание могло не работать на одном компьютере и работать на другом:
# там, где deno случайно стоял в системе, всё работало. Каскад node/bun
# (в downloader.py) остаётся запасным вариантом.
yt-dlp[default]>=2026.8.19
deno>=2.6.6

# ffmpeg-обёртка для Python (для медиа-операций)
ffmpeg-python==0.2.0

# HTTP-клиент (для yt-dlp fallback каналов)
# [Security bump 2026-10-19]: CVE-2024-47081 (.netrc leak) + CVE-2026-25645
# (insecure temp file) — 2.33.0 минимальная версия, закрывающая оба.
requests==2.33.0

# TLS-impersonation — обход блокировок YouTube (HTTP 429, bot-защита)
curl-cffi==0.15.0

# Переводчик (engine_server.py импортирует Translator, даже если веб-интерфейс
# не использует перевод — это нужно для импорта без ошибки)
deep-translator==1.11.4

# ─── Конвертация документов и изображений (движок ≥ 1.3.0) ───
# DOCX (python-docx), PDF-чтение (pypdf), PDF-рендер (reportlab),
# PDF→DOCX (pdf2docx), DOCX→PDF при наличии MS Word (docx2pdf),
# изображения (Pillow). Без них конвертер документов даёт ImportError.
# Pillow 11.3.0 — первая версия с готовыми wheels для Python 3.14:
# на 11.1.0 pip пытался собирать Pillow из исходников и падал без zlib.
python-docx==1.1.2
# [Security bump 2026-10-19, OSV-аудит]: pypdf 5.1.0 — 97 CVE (DoS на
# зловредных PDF: бесконечные циклы/исчерпание RAM). Все закрыты в 6.x
# → 6.21.1. Wheels не нужны (pure-python). Наши вызовы (PdfReader) не менялись.
pypdf==6.21.2
reportlab==4.2.5
pdf2docx==0.5.8
docx2pdf==0.1.8
# [Security bump 2026-10-19, OSV-аудит]: Pillow 11.x — 34 CVE (парсеры
# изображений). Минимальная версия, закрывающая все — 12.3.0; wheels
# cp310–cp314 есть на всех платформах (проверено PyPI).
Pillow==12.3.0

# torch тянется автоматически через openai-whisper. Фиксируем мин. версию
# для воспроизводимости. CPU-only на macOS/Linux/Windows.
# Если у вас GPU — установите torch с поддержкой CUDA вручную:
#   pip install torch --index-url https://download.pytorch.org/whl/cu121
torch>=2.0.0
